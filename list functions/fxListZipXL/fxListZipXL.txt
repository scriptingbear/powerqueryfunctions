/*  fxListZipXL()

    A function to "zip" lists of primitive values (numbers, text, logicals etc.),
    For ordinary lists, optionally populate excess list items (when the lists are of different sizes) with
    something other than null, which is the default behavior of List.Zip(). Also, provide an option to 
    zip list elements up to the size of the smallest list

    Parameters:
                items_to_zip - a non empty list of non empty lists of primitive data types


*/

(items_to_zip as list, optional default_value as any, optional trim_lists as logical) as any =>
let
    //Validations for regular zipping
    items_to_zip_ok = (items_to_zip <> {}) 
                      and 
                      (List.Count(items_to_zip) > 1) 
                      and   
                      ( 
                        //Ensure each item is a non empty list and all elements of inner lists are not themselves
                        //lists or records
                        List.AllTrue(
                                        List.Transform(
                                                        items_to_zip, 
                                                        each (_ is list) 
                                                        and (_ <> {})
                                                        and (
                                                               List.AllTrue(
                                                                   List.Transform(_, 
                                                                        each List.Contains({type text,
                                                                                       type number,
                                                                                       type date,
                                                                                       type datetime,
                                                                                       type duration,
                                                                                       type logical,
                                                                                       type null},
                                                                                       Value.Type(_))
                                                                                    )
                                                               )
                                                        )
                                                     )
                                    )
                      ),
    default_value_ok = List.Contains({type null, type number, type text, type logical}, Value.Type(default_value)),
    //Caution: Originally, the next step was labeled "trim_lists", just like the parameter name
    //This can sometimes cause issues, especially if this function is used in a table
    //Unlike other languages, M Code doesn't seem to like assigning expressions to 
    //function parameters. So in this case, I made a slight change to "trim_lists_mode".
    //Doing that prevented any further errors, even when this function is invoked in a custom column
    trim_lists_mode = trim_lists ?? false,

    smallest_list_size = List.Min(List.Transform(items_to_zip, each List.Count(_))),
    fxTrimAllLists = (list_of_lists as list, list_size as number) as list =>
        List.Transform(list_of_lists, each List.FirstN(_, list_size)),

    zipped_list = if items_to_zip_ok and default_value_ok then 
                    let 
                        trimmed_lists = if trim_lists_mode then fxTrimAllLists(items_to_zip, smallest_list_size)
                                        else 
                                           items_to_zip,
                        zip_trimmed_lists = List.Zip(trimmed_lists),
                        add_default_value = 
                            if default_value_ok and (default_value <> null) and (not trim_lists_mode) then 
                                //There is a risk that original null values, i.e. not generated by List.Zip() will be overwritten
                                //with default_value
                                List.Transform(zip_trimmed_lists, each List.ReplaceValue(_, null, default_value, Replacer.ReplaceValue))
                            else  
                                zip_trimmed_lists
                    in  
                        add_default_value
                 else 
                    {} 
in
    zipped_list
