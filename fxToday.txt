/*  fxToday()

    A simple wrapper function that mimic's Excels TODAY() function, with an 
    optional format specifier to return the date as a string. It can be useful
    in constructing strings where the current date needs to be inserted, like in a report
    header of footer.
    It's easier than using Date.ToTex(Date.From(DateTime.LocalNow()), [Format = "mm/dd/yy"]) or
    similar.

    [NEW CODE: 07/28/2026] - Simulate function overloading by allowing first and only parameter to
                             have any of the following values:
                             null (missing) -> return a date object for today's date
                             string -> if valid, return a date as string with specified format

                             >>A separate date_format parameter is no longer needed, making the 
                             function more user friendly.<<

    Parameters: function_mode - null, false, string
                [DEPRECATED]: date_format - string with valid date format codes; PQ throws error for invalid codes 

    Dependencies: fxDataTypeAsText()
*/

(optional function_mode as any) as any =>

let
    /*  If function_mode is null, false, or missing - return date object with today's date
        If function mode is a string, return date formatted with specified date format, if valid
        If function mode is some other data type, throw an error

    */
    current_date = Date.From(DateTime.LocalNow()),
    data_types = {
                    (function_mode = null),
                    (List.Contains({false, true}, function_mode)),
                    (function_mode = ""),
                    (function_mode is text),
                    true //fallback case input is invalid, 
                         //e.g. {} or [] data type
                },
    data_types_index = List.PositionOf(data_types, true),

    return_values = {
                        current_date,               //null
                        current_date,               //false or true
                        Date.ToText(current_date), //"" -> return system date as string in default format, e.g. "28-07-2026"
                        Date.ToText(current_date, [Format = function_mode]) //-> return system date as string in specified format, e.g.                                                                 "28-07-2026"
                                                                     
                    },



    output = try return_values{data_types_index} 
                otherwise error Error.Record("Invalid input", "The input value is not null, logical, """", or a valid date format as text")
        
in
    output
