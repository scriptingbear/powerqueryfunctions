/*  fxTextFromList()

    A function to return a text representation of  a list of primitive data type, 
    similar to fxTextFromRecord().

    Parameters: input_list - must not be empty and must not contain structured data. Null values
                             will be represented by the string "null" in the output

                 display_vertically - optionally display record with one line per list item
    
    [NEW CODE: 08/19/2026] Corrected bug where null elements in list were being ignored, insted
                           of being represented by the text literal "null".
    [NEW CODE: 08/26/2026] Now able to represent structured values like lists, records, and tables
                           in a very simplified and limited way.
        

*/


(input_list as list, optional display_vertically as logical) as text =>

let 
    cr_lf = "#(cr)" & "#(lf)",
    tab = "#(tab)",
    bad_value = "?val!",  //can't be given a text representation (easily) 
    is_valid_list = input_list <> {},
    use_line_breaks = display_vertically ?? false,
    fxRecordAsText = (record_object as record) as text =>
        let  
            start_bracket = "[",
            end_bracket = "]",
            field_names = Record.FieldNames(record_object),
            field_values = Record.FieldValues(record_object),
            add_quotes = List.Transform(field_values, each if Value.Is(_, type text) then """" & _ & """" else _),
            safe_field_values = List.Transform(add_quotes, each try Text.From(_) otherwise bad_value),
            zipped_items = List.Zip({field_names, safe_field_values}),
            formatted_items = Text.Combine(List.Transform(zipped_items, each Text.Combine(_, " = ")), ", "),
            formatted_record = Text.Combine({start_bracket, formatted_items, end_bracket}, "")
        in 
            formatted_record, 

    fxListAsText = (list_object as list) as text =>
            let 
                start_brace = "{",
                end_brace = "}",
                add_quotes = List.Transform(list_object, each if Value.Is(_, type text) then """" & _ & """" else _),
                safe_items = List.Transform(add_quotes, each try Text.From(_) otherwise bad_value),
                formatted_safe_items = Text.Combine(safe_items, ", "),
                formatted_list = Text.Combine({start_brace, formatted_safe_items, end_brace}, "")
            in 
                formatted_list,
    
    fxTableAsText = (table_object as table) as text =>
            let 
                start_bracket = "[",
                end_bracket = "]",
                column_names = Table.ColumnNames(table_object),
                table_header = Text.Combine(column_names, "|"),
                formatted_table = Text.Combine({start_bracket, table_header, end_bracket})
            in 
                formatted_table,

    list_as_text = if is_valid_list then
                        let 
                            delimited_text_items_in_list = 
                                List.Accumulate(
                                                input_list,
                                                {},
                                                (state, current) =>
                                                    let 
                                                        //Delimit text values with quotes
                                                        is_text = current is text,
                                                        is_record = current is record,
                                                        is_table = current is table,
                                                        is_list = current is list,
                                                        add_quotes =
                                                            if is_text then 
                                                                """" & current & """"
                                                            else if current is null then 
                                                                "null"
                                                            else if is_record then 
                                                                fxRecordAsText(current)
                                                            else if is_table then 
                                                                fxTableAsText(current)
                                                            else if is_list then 
                                                                fxListAsText(current)
                                                            else
                                                                try Text.From(current) otherwise bad_value,
                                                        updated_element = state & {(if use_line_breaks then tab else "") & add_quotes}
                                                    in  
                                                        updated_element
                                                    
                                                ),
                            add_braces = "{" 
                                & (if use_line_breaks then cr_lf else "")
                                & Text.Combine(delimited_text_items_in_list, (if use_line_breaks then "," & cr_lf else ", ")) 
                                & (if use_line_breaks then cr_lf else "")
                                & "}"
                        in 
                            add_braces
                    else  
                        "{}"

in 
    list_as_text
